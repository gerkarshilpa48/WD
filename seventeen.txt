    <script>

        function validateForm() {

            let name =
            document.forms["shopForm"]["name"].value;

            let email =
            document.forms["shopForm"]["email"].value;

            let mobile =
            document.forms["shopForm"]["mobile"].value;

            let password =
            document.forms["shopForm"]["password"].value;

            let emailPattern =
            /^[^ ]+@[^ ]+\.[a-z]{2,3}$/;

            if (name == "") {

                alert("Please enter name");

                return false;
            }

            if (!email.match(emailPattern)) {

                alert("Enter valid email");

                return false;
            }

            if (mobile.length != 10 || isNaN(mobile)) {

                alert("Enter valid 10 digit mobile number");

                return false;
            }

            if (password.length < 6) {

                alert("Password must contain minimum 6 characters");

                return false;
            }

            return true;
        }

    </script>
