    <script>

        function calculateFine() {

            let days = parseInt(document.getElementById("days").value);

            let fine = 0;

            switch(true) {

                case (days <= 5):
                    fine = days * 2;
                    break;

                case (days <= 10):
                    fine = days * 5;
                    break;

                default:
                    fine = days * 10;
                    break;
            }

            document.getElementById("result").innerHTML =
            "Book Fine = ₹" + fine;
        }
