    <script>

        function validateForm() {

            let name = document.forms["eventForm"]["name"].value;
            let email = document.forms["eventForm"]["email"].value;

            if (name == "") {
                alert("Please enter your name");
                return false;
            }

            if (email == "") {
                alert("Please enter email");
                return false;
            }

            alert("Event Registration Successful!");

            return true;
        }
