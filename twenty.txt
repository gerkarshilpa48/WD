    <script>

        function validateForm() {

            let name =
            document.forms["hostelForm"]["name"].value;

            let mobile =
            document.forms["hostelForm"]["mobile"].value;

            if (name == "") {

                alert("Please enter student name");

                return false;
            }

            if (mobile.length != 10 || isNaN(mobile)) {

                alert("Enter valid 10 digit mobile number");

                return false;
            }

            return true;
        }
