<script>

    function showTime() {

        let now = new Date();

        let hours = now.getHours();

        let minutes = now.getMinutes();

        let seconds = now.getSeconds();

        // 24 Hour Format

        let h24 = hours < 10 ? "0" + hours : hours;

        let m24 = minutes < 10 ? "0" + minutes : minutes;

        let s24 = seconds < 10 ? "0" + seconds : seconds;

        let time24 = h24 + ":" + m24 + ":" + s24;

        document.getElementById("clock24").innerHTML =
        time24;

        // 12 Hour Format

        let period = hours >= 12 ? "PM" : "AM";

        let h12 = hours % 12;

        h12 = h12 ? h12 : 12;

        h12 = h12 < 10 ? "0" + h12 : h12;

        let m12 = minutes < 10 ? "0" + minutes : minutes;

        let s12 = seconds < 10 ? "0" + seconds : seconds;

        let time12 =
        h12 + ":" + m12 + ":" + s12 + " " + period;

        document.getElementById("clock12").innerHTML =
        time12;

        // Current Date

        let currentDate = now.toDateString();

        document.getElementById("date").innerHTML =
        currentDate;
    }

    // Update Every Second

    setInterval(showTime, 1000);

    showTime();

</script>
