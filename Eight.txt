    <script>

        function validateForm() {

            let pname = document.forms["appointForm"]["pname"].value;
            let mobile = document.forms["appointForm"]["mobile"].value;

            if (pname == "") {
                alert("Please enter patient name");
                return false;
            }

            if (mobile == "") {
                alert("Please enter mobile number");
                return false;
            }

            if (mobile.length != 10 || isNaN(mobile)) {
                alert("Enter valid 10 digit mobile number");
                return false;
            }

            return true;
        }
